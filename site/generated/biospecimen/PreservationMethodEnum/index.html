    name: PreservationMethodEnum
title: preservation_method
from_schema: https://w3id.org/htan/preservation_method_enum
permissible_values:
  Cryopreserved:
    text: Cryopreserved
    description: Preservation of cells, tissues, organs, or embryos by storage at
      low temperatures.
  Cytospin Slide:
    text: Cytospin Slide
    description: A microscope slide prepared by depositing cells directly onto a slide
      by centrifugation. The cytocentrifuge spins at an angle, at low speeds, allowing
      the cells to adhere to the slide in a monolayer format.
  Fixation:
    text: Fixation
    description: The chemical preservation of a substance to maintain structural and/or
      molecular features of that substance as they exist in the living entity.
  Fresh:
    text: Fresh
    description: Tissue which has not been exposed to a fixative solution.
  Fresh Dissociated:
    text: Fresh Dissociated
    description: A tissue preparation process that takes fresh tissue and dissociates
      it into single cell suspensions.
  Fresh Dissociated and Single Cell Sorted:
    text: Fresh Dissociated and Single Cell Sorted
    description: A tissue preparation process that separates fresh dissociated tissue
      cells into cell populations by single cell sorting.
  Fresh Dissociated and Single Cell Sorted into Plates:
    text: Fresh Dissociated and Single Cell Sorted into Plates
    description: A tissue preparation process that takes dissociated, sorted cells
      and distributes them into cell propagation plates.
  Frozen:
    text: Frozen
    description: A specimen that has been subjected to and immobilized by severe cold.
  Not Applicable:
    text: Not Applicable
    description: Determination of a value is not relevant in the current context.
  Not Reported:
    text: Not Reported
    description: Not provided or available.
  Refrigerated:
    text: Refrigerated
    description: A specimen that has been kept or preserved at a low temperature in
      a refrigerator.
  Refrigerated Vacuum Chamber:
    text: Refrigerated Vacuum Chamber
    description: A device designed to maintain a vacuum at a below-ambient temperature.
  Snap Frozen:
    text: Snap Frozen
    description: To freeze rapidly so as to preserve structure and prevent ice crystal
      formation.
  Unknown:
    text: Unknown
    description: Not known, not observed, not recorded, or refused.
